[On Server]-------------------------
[root@master ~]# dnf install nfs-utils -y
[root@master ~]# mkdir /nfsshare

[root@master ~]# firewall-cmd --add-service=nfs --permanent
success

[root@master ~]# firewall-cmd --reload
success

[root@master ~]# cd /nfsshare

[root@master nfsshare]# touch file{1..10}.txt

[root@master nfsshare]# ls
file10.txt  file2.txt  file4.txt  file6.txt  file8.txt
file1.txt   file3.txt  file5.txt  file7.txt  file9.txt

[root@master nfsshare]# vim /etc/exports
          
[root@master nfsshare]# systemctl restart nfs-server
[root@master nfsshare]# systemctl enable nfs-server
